    ```python
    """Integration tests for Phase 6 — Compliance module with real PDFs."""

    import pytest
    from pathlib import Path
    from tests.conftest import PDF_DIR

    PAR1_DIR = PDF_DIR / "Par1"  # Adjust if docs are in root dir

    @pytest.mark.integration
    class TestPar1ComplianceIntegration:
        """Par 1: Ruan (Administrador) signs atestado -> needs Declaracao -> Alexandre's Declaracao present."""

        def test_par1_declaracao_necessaria(self):
            """Ruan (Administrador, engenheiro_habilitado=False) triggers Declaracao requirement."""
            # Import here to avoid module-level OpenAI client creation issues
            from crea_cat_ai import ingestion, analysis, matching, compliance
            from crea_cat_ai.ingestion import TipoDocumento

            # Check if test PDFs exist
            if not PDF_DIR.exists():
                pytest.skip(f"PDF directory not found: {PDF_DIR}")

            # Load and analyze
            pacote = ingestion.carregar_pacote(PDF_DIR)
            assert pacote.art is not None, "ART not found in test docs"
            assert pacote.atestado is not None, "Atestado not found in test docs"

            resultados_analise = analysis.analisar_pacote(pacote)

            art_analise = resultados_analise.get(TipoDocumento.ART.value)
            atestado_analise = resultados_analise.get(TipoDocumento.ATESTADO.value)
            assert art_analise is not None
            assert atestado_analise is not None

            resultado_cruzamento = matching.cruzar_documentos(art_analise, atestado_analise)

            # Run compliance
            resultado = compliance.verificar_compliance(
                pacote=pacote,
                resultados_analise=resultados_analise,
                resultado_cruzamento=resultado_cruzamento,
            )

            # Par 1: Ruan is Administrador -> Declaracao should be required
            assert resultado.declaracao_necessaria is True, "Ruan (Administrador) should trigger Declaracao requirement"

            # Par 1: Alexandre's Declaracao file should be found in complementares
            declaracao_found = any(
                d.tipo == TipoDocumento.DECLARACAO for d in pacote.complementares
            )
            if declaracao_found:
                assert resultado.declaracao_encontrada is True, "Alexandre's Declaracao should be found"

            # Compliance should have verificacoes
            assert len(resultado.verificacoes) > 0, "Should have at least one verificacao"
    ```
